# Dependencies for the MCP voice summary server.
#
# mcp >=2.0: from version 2.0 onwards, FastMCP was renamed to MCPServer.
# edge-tts: Azure neural voices (edge engine, requires internet).
# langdetect: language detection, only used if you set VOICE_LANGUAGE=auto.
#             Optional: without it the server still works and just logs a
#             warning when auto detection is requested.
#
# The sapi5 engine is also optional. You only need it for offline speech:
#
#   Windows: pip install pyttsx3 pywin32 comtypes
#   Linux:   pip install pyttsx3  plus the system engine:
#            sudo apt install espeak-ng libespeak-ng1
#   macOS:   pip install pyttsx3
#
# pyttsx3 is therefore not installed by default: the edge engine does not need
# it, which keeps the project cross-platform with no system dependencies.

mcp>=2.0,<3
edge-tts>=7.0
langdetect>=1.0.9

# Optional sapi5 engine. Uncomment if you plan to use it:
# pyttsx3>=2.90
# pywin32>=306; sys_platform == "win32"
# comtypes>=1.2; sys_platform == "win32"
